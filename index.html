async function generate() { const topics = Array.from(document.querySelectorAll('.chip.active')).map(c => c.innerText); if(!topics.length) return alert("Select a topic!"); const status = document.getElementById('status'); const output = document.getElementById('output'); status.innerText = "🛰️ Handshaking with AI..."; try { // We use 'Llama-3.2-1B' - it's newer, faster, and very stable for web calls const response = await fetch("https://api-inference.huggingface.co/models/meta-llama/Llama-3.2-1B-Instruct", { method: "POST", mode: "cors", // Explicitly telling the browser to use CORS headers: { "Authorization": `Bearer ${localStorage.getItem('dg_key').trim()}`, "Content-Type": "application/json" }, body: JSON.stringify({ inputs: `<|begin_of_text|><|start_header_id|>user<|end_header_id|>\n\nWrite a short, fun teen news report about ${topics.join(", ")} with emojis.<|eot_id|><|start_header_id|>assistant<|end_header_id|>\n\n`, parameters: { max_new_tokens: 300 } }), }); if (!response.ok) { const errData = await response.json(); throw new Error(errData.error || "Server Error"); } const result = await response.json(); // Handling different ways the AI sends back text let text = result[0]?.generated_text || result.generated_text || "No response."; if (text.includes("<|end_header_id|>")) text = text.split("<|end_header_id|>").pop(); output.innerHTML = text.replace(/\n/g, '
'); output.style.display = "block"; status.innerText = "✨ Success!"; } catch (e) { console.error("CORS/Fetch Error:", e); status.innerText = "❌ Connection Blocked by Browser"; alert("CRITICAL: Your browser or an extension is blocking the AI. Try Incognito mode!"); } }
